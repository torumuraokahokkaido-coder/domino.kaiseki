import tempfile import cv2 import numpy as np import pandas as pd import streamlit as st import plotly.express as px import plotly.graph_objects as go # 画面を広く使う設定 st.set_page_config(layout="wide", page_title="ドミノ運動解析 Pro") st.title("ドミノ運動解析アプリ 🎲 (伝播速度 ＆ 個別倒伏速度対応)") st.write("最大30個の赤いマーカーを追跡し、ドミノの変位、倒伏速度、波の伝播速度を解析します。") # ========================================== # サイドバーの設定パネル # ========================================== st.sidebar.header("1. 解析パラメータ設定") min_area = st.sidebar.slider("マーカーの最小サイズ (ピクセル)", 10, 500, 30, help="このサイズ以上の赤い印をドミノとして認識します。") movement_threshold = st.sidebar.slider("「動き出し」判定の変位 (ピクセル)", 1, 30, 5, help="このピクセル数以上動いたら「倒れ始めた」と判定し、波の伝播速度の計算に使います。") st.sidebar.header("2. 速度計算ノイズ除去") smooth_window = st.sidebar.slider("速度の平滑化（移動平均フレーム数）", 1, 10, 3, help="数値が大きいほどグラフのギザギザが滑らかになります。") st.sidebar.header("3. スケール設定 (ピクセル → mm)") st.sidebar.write("プレビュー画面で2点のX座標を読み取り、実際の長さを入力してください。") pixel_x1 = st.sidebar.number_input("点1の X座標 (px)", value=100) pixel_x2 = st.sidebar.number_input("点2の X座標 (px)", value=300) real_length_mm = st.sidebar.number_input("実際の長さ (mm)", value=50.0) # ピクセルからmmへの変換係数を計算 if abs(pixel_x2 - pixel_x1) > 0: pixel_to_mm = real_length_mm / abs(pixel_x2 - pixel_x1) else: pixel_to_mm = 1.0 # ========================================== # メイン画面：動画のアップロードと解析 # ========================================== uploaded_file = st.file_uploader("ドミノの動画を選択 (MP4 / MOV)", type=["mp4", "mov"]) if uploaded_file is not None: # アップロードされた動画を一時ファイルに保存 tfile = tempfile.NamedTemporaryFile(delete=False) tfile.write(uploaded_file.read()) # 解析を一度だけ実行し、結果をキャッシュする関数 @st.cache_data(show_spinner=False) def analyze_video(video_path, min_area, max_dominoes=30): cap = cv2.VideoCapture(video_path) fps = cap.get(cv2.CAP_PROP_FPS) records = [] frame_images = [] frame_count = 0 while cap.isOpened(): ret, frame = cap.read() if not ret: break frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV) # HSV色空間で赤色を抽出（0付近と180付近の2つの領域） lower_red1, upper_red1 = np.array([0, 100, 100]), np.array([10, 255, 255]) lower_red2, upper_red2 = np.array([170, 100, 100]), np.array([180, 255, 255]) mask = cv2.inRange(hsv, lower_red1, upper_red1) + cv2.inRange(hsv, lower_red2, upper_red2) # ノイズ除去と輪郭抽出 contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE) valid_centers = [] for c in contours: if cv2.contourArea(c) >= min_area: M = cv2.moments(c) if M["m00"] != 0: cx = int(M["m10"] / M["m00"]) cy = int(M["m01"] / M["m00"]) valid_centers.append((cx, cy)) # X座標の小さい順（左から右）に並べ替え、最大30個に制限 valid_centers.sort(key=lambda pt: pt[0]) valid_centers = valid_centers[:max_dominoes] # プレビュー画像に認識した点と番号を描画 preview_img = frame_rgb.copy() for idx, (cx, cy) in enumerate(valid_centers): cv2.circle(preview_img, (cx, cy), 5, (0, 255, 0), -1) # 緑の円を描画 cv2.putText(preview_img, str(idx+1), (cx-10, cy-15), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2) # データを記録 records.append({ "Frame": frame_count, "Time (s)": frame_count / fps, "Domino ID": f"Domino {idx + 1:02d}", "Px X": cx, "Px Y": cy, }) frame_images.append(preview_img) frame_count += 1 cap.release() return pd.DataFrame(records), frame_images, fps with st.spinner('動画を解析中...（数十秒かかる場合があります）'): raw_df, frame_images, fps = analyze_video(tfile.name, min_area) if not raw_df.empty: st.success("解析完了！") # Plotlyで画像を表示（マウスオーバーで座標取得可能） fig_img = px.imshow(frame_images[frame_slider]) fig_img.update_layout(margin=dict(l=0, r=0, b=0, t=0)) st.plotly_chart(fig_img, use_container_width=True) # ========================================== # 2. 物理データ（変位・速度）の計算 # ========================================== df = raw_df.copy() # ピクセルをmmに変換 df["Position X (mm)"] = df["Px X"] * pixel_to_mm df["Position Y (mm)"] = df["Px Y"] * pixel_to_mm # 変位の計算（最初のフレームの位置を0とする） df["Disp X (mm)"] = df.groupby("Domino ID")["Position X (mm)"].transform(lambda x: x - x.iloc[0]) # Y軸は下方向がプラスなので、そのまま引いて「落下量」とする df["Disp Y (mm)"] = df.groupby("Domino ID")["Position Y (mm)"].transform(lambda x: x - x.iloc[0]) df["Total Disp (mm)"] = np.sqrt(df["Disp X (mm)"]**2 + df["Disp Y (mm)"]**2) # ドミノ個体の落下速度（Y軸方向）の計算 dt = 1.0 / fps df["Raw Velocity Y (mm/s)"] = df.groupby("Domino ID")["Disp Y (mm)"].diff() / dt # ノイズ除去（移動平均） df["Velocity Y (mm/s)"] = df.groupby("Domino ID")["Raw Velocity Y (mm/s)"].transform( lambda x: x.rolling(window=smooth_window, min_periods=1).mean() ) # ========================================== # 3. 解析グラフの表示（タブ切り替え） # ========================================== st.header("2. 解析グラフ (凡例クリックで表示/非表示)") tab1, tab2, tab3 = st.tabs(["① ドミノの変位", "② 個別の倒伏速度 (v-t)", "③ 波の伝播速度"]) # --- タブ①: 変位 --- with tab1: st.write("各ドミノが時間とともにどれだけ落下（Y軸方向に変位）したかを示します。") fig_disp = px.line(df, x="Time (s)", y="Disp Y (mm)", color="Domino ID", title="各ドミノのY軸方向の変位（落下量）", markers=True) st.plotly_chart(fig_disp, use_container_width=True) # --- タブ②: 各ドミノ個体の倒れる速度 --- with tab2: st.write("各ドミノが倒れ込む際のスピー​​ド変化です。衝突時の急減速などが観察できます。") fig_ind_v = px.line(df, x="Time (s)", y="Velocity Y (mm/s)", color="Domino ID", title="ドミノ個体の倒伏速度 (mm/s)", markers=True) st.plotly_chart(fig_ind_v, use_container_width=True) # 最高速度の一覧表 max_v_df = df.groupby("Domino ID")["Velocity Y (mm/s)"].max().reset_index() max_v_df.columns = ["Domino ID", "最大倒伏速度 (mm/s)"] st.dataframe(max_v_df.T) # --- タブ③: ドミノ間の伝播速度 --- with tab3: st.write("ドミノが次々と倒れていく「波」のスピード（伝播速度）です。") start_times = [] domino_ids = sorted(df["Domino ID"].unique()) #
